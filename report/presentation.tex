\begin{frame}{Background \& Problem Identification}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{The Challenge in Aquaponics:}
      \begin{itemize}
        \item Highly delicate balance between aquaculture and hydroponics.
        \item High vulnerability to silent system crashes (e.g., nitrogen spikes, oxygen dips).
        \item Traditional nutrient cycles miss critical plant macro/micronutrients.
      \end{itemize}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \textbf{Key Operational Bottlenecks:}
      \begin{itemize}
        \item \textbf{Ammonia \& Nitrite Toxicity:} Rapid fish mortality events.
        \item \textbf{Nutrient Deficiencies:} Reduced plant yield due to incomplete fish waste breakdown.
        \item \textbf{Manual Monitoring Risk:} High human labor cost and delayed responses to chemical changes.
      \end{itemize}
    \end{column}
  \end{columns}
  
  \vspace{0.5cm}
  \begin{block}{Problem Statement}
    Traditional aquaponics lacks real-time predictive safeguards, leading to financial loss, unstable yields, and high operational overhead.
  \end{block}
\end{frame}

\begin{frame}{Technical Feasibility}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Robust Multi-Tier Hardware}
      \begin{itemize}
        \item High-performance local compute (Raspberry Pi 5) for computer vision processing.
        \item Real-time industrial microcontrollers (STM32H7 / ESP32) for low-latency actuator/sensor control.
      \end{itemize}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \textbf{Reliable ROS2 Architecture}
      \begin{itemize}
        \item Modular node communication for sensors, AI modules, and actuators.
        \item Failsafe routines built into embedded controllers to maintain water pumps/aeration even during network downtime.
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.4cm}
  \begin{block}{Safety Redundancies}
    Physical bypass valves, hardware-level emergency aeration triggers, and sensor sanity checks prevent systemic failures.
  \end{block}
